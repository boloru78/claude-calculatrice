# Parcours de tous les écrans de l'app, avec captures (make screenshots).
# Lancé aussi par make check : il échoue si un texte sort de l'écran.
# Le faux ESP32 répond toujours pareil (SIM_ESP32=faux). Syntaxe des
# scripts : sim/README.md
DELAY:150
WAIT:12                         # l'horloge tourne
SHOT:accueil

# Une question, puis une deuxième plus longue.
EXE
TEXTE:C'est_quoi_pi_?
SHOT:clavier
EXIT EXE                        # on referme et rouvre : le texte reste
TEXTE:|                         # ↵ : envoi
WAIT:5
SHOT:reponse
EXE
TEXTE:Et_la_dérivée_de_x^2_?|
WAIT:5
SHOT:reponse_longue
UP UP UP                        # défilement
SHOT:defilement
DOWN DOWN DOWN

# Les pages du clavier et les majuscules.
EXE
SHIFT
SHOT:clavier_majuscule
UP EXE                          # touche de page : 123
SHOT:clavier_123
EXE                             # éà
SHOT:clavier_accents
EXE                             # abc
RIGHT DOWN LEFT DEL AC
EXIT

# Menu, Wi-Fi, nouvelle conversation.
EXIT
SHOT:menu
DOWN DOWN EXE                   # Wi-Fi
SHOT:wifi
EXE                             # Chercher les réseaux
SHOT:reseaux
DOWN EXE                        # Ecole-Personnel : nom d'utilisateur
SHOT:wifi_utilisateur
TEXTE:eleve@ecole.ca|
SHOT:wifi_mot_de_passe
TEXTE:mauvais|                  # refusé par le faux ESP32
SHOT:wifi_refuse
EXIT DOWN DOWN EXE EXE          # de nouveau la liste
DOWN DOWN DOWN DOWN DOWN DOWN   # le dernier : défilement
SHOT:reseaux_defilement
UP UP UP UP UP UP               # MaisonWifi : mot de passe connu
EXE
SHOT:wifi_connu
EXE                             # Se connecter
SHOT:wifi_connecte
EXIT DOWN DOWN EXE DOWN EXE     # Wi-Fi, réseau caché
TEXTE:Garage|
SHOT:wifi_securite
EXE                             # Mot de passe
TEXTE:secret123|
EXIT DOWN DOWN EXE EXE DOWN DOWN DOWN DOWN EXE  # Labo-Securise
SHOT:wifi_certificat
EXE EXIT EXIT
EXIT DOWN EXE                   # Nouvelle conversation
SHOT:nouvelle

# L'ESP32 débranché, puis sans Wi-Fi.
ESP32:absent
WAIT:110                        # 11 s : nouvelle demande d'état
SHOT:sans_esp32
EXE
TEXTE:Allo_?|
SHOT:erreur_esp32
ESP32:sanswifi
WAIT:110
EXE
TEXTE:Allo_?|
SHOT:sans_wifi
ESP32:faux
WAIT:110

# Aide complète, puis Quitter.
EXIT DOWN DOWN DOWN EXE
SHOT:aide
RIGHT RIGHT RIGHT RIGHT RIGHT RIGHT LEFT RIGHT
EXE                             # EXE sur la dernière page : retour
EXIT DOWN DOWN DOWN DOWN EXE
